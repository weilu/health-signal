# health-signal — Stage 2 (PR #4): App-managed authentication — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Gate the app behind app-managed authentication — a pluggable `AuthProvider` with a `LocalAccountsProvider` (argon2-hashed accounts from env/Connect secrets, signed-cookie sessions), login/logout/me routes, and a `require_auth` dependency — so nothing but health and the login flow is reachable unauthenticated.

**Architecture:** `create_app(config, auth_provider=None)` wires an `AuthProvider`. `LocalAccountsProvider` verifies credentials (argon2) and issues/reads a signed session cookie (itsdangerous). A `require_auth` dependency protects `/api/*`; the SPA catch-all redirects unauthenticated users to `/login`. `/healthz` stays public.

**Tech Stack:** FastAPI, Pydantic v2, argon2-cffi (password hashing), itsdangerous (signed cookies), python-multipart (form POST); Starlette `TestClient`.

**Spec:** `docs/specs/2026-09-23-reusable-surveillance-dashboard-design.md` (§3 gating rationale, §7.2 AuthProvider, §8 routes).

- [ ] **Step 4: Write `src/health_signal/auth.py`**

```python
import json
import os
from typing import Protocol

from argon2 import PasswordHasher
from argon2.exceptions import VerifyMismatchError
from fastapi import Request, Response
from itsdangerous import BadSignature, SignatureExpired, URLSafeTimedSerializer
from pydantic import BaseModel

_ph = PasswordHasher()
_COOKIE = "hs_session"


class User(BaseModel):
    email: str


def hash_password(password: str) -> str:
    return _ph.hash(password)


class AuthProvider(Protocol):
    def authenticate(self, email: str, password: str) -> "User | None": ...
    def current_user(self, request: Request) -> "User | None": ...
    def issue_session(self, response: Response, user: User) -> None: ...
    def clear_session(self, response: Response) -> None: ...


class LocalAccountsProvider:
    def __init__(self, accounts: dict[str, str], secret_key: str, max_age_s: int = 43200):
        self._accounts = accounts
        self._max_age = max_age_s
        self._serializer = URLSafeTimedSerializer(secret_key, salt="hs-session")

    def authenticate(self, email: str, password: str) -> User | None:
        stored = self._accounts.get(email)
        if stored is None:
            return None
        try:
            _ph.verify(stored, password)
        except VerifyMismatchError:
            return None
        return User(email=email)

    def issue_session(self, response: Response, user: User) -> None:
        token = self._serializer.dumps({"email": user.email})
        response.set_cookie(
            _COOKIE, token, max_age=self._max_age,
            httponly=True, secure=True, samesite="lax",
        )

    def clear_session(self, response: Response) -> None:
        response.delete_cookie(_COOKIE)

    def current_user(self, request: Request) -> User | None:
        token = request.cookies.get(_COOKIE)
        if not token:
            return None
        try:
            data = self._serializer.loads(token, max_age=self._max_age)
        except (BadSignature, SignatureExpired):
            return None
        return User(email=data["email"])


def from_env(max_age_s: int = 43200) -> LocalAccountsProvider:
    secret = os.environ.get("HEALTH_SIGNAL_SECRET_KEY")
    if not secret:
        raise RuntimeError("HEALTH_SIGNAL_SECRET_KEY is required for authentication.")
    accounts = json.loads(os.environ.get("HEALTH_SIGNAL_ACCOUNTS", "{}"))
    return LocalAccountsProvider(accounts=accounts, secret_key=secret, max_age_s=max_age_s)
```

from starlette.requests import Request as StarletteRequest


def _request_with_cookie(name, value):

```python
from fastapi import Depends, FastAPI, Form, HTTPException, Request, Response
from fastapi.responses import HTMLResponse, RedirectResponse, FileResponse

from . import auth as _auth
from .auth import AuthProvider, User

_LOGIN_FORM = (
    "<!doctype html><meta charset='utf-8'><title>Sign in</title>"
    "<h1>Sign in</h1><form method='post' action='/login'>"
    "<label>Email <input name='email' type='email'></label> "
    "<label>Password <input name='password' type='password'></label> "
    "<button type='submit'>Sign in</button>{error}</form>"
)


def create_app(config: Config, auth_provider: AuthProvider | None = None) -> FastAPI:
    provider = auth_provider or _auth.from_env()
    app = FastAPI(title=config.site.title)

    def require_auth(request: Request) -> User:
        user = provider.current_user(request)
        if user is None:
            raise HTTPException(status_code=401, detail="Not authenticated")
        return user

    @app.get("/healthz")
    def healthz() -> dict:
        return {"status": "ok", "schema_version": config.schema_version}

    @app.get("/login", response_class=HTMLResponse)
    def login_form() -> HTMLResponse:
        return HTMLResponse(_LOGIN_FORM.format(error=""))

    @app.post("/login")
    def login(response: Response, email: str = Form(...), password: str = Form(...)):
        user = provider.authenticate(email, password)
        if user is None:
            return HTMLResponse(_LOGIN_FORM.format(error="<p>Invalid credentials.</p>"), status_code=401)
        redirect = RedirectResponse("/", status_code=303)
        provider.issue_session(redirect, user)
        return redirect

    @app.post("/logout")
    def logout():
        redirect = RedirectResponse("/login", status_code=303)
        provider.clear_session(redirect)
        return redirect

    @app.get("/api/me")
    def me(user: User = Depends(require_auth)) -> dict:
        return {"email": user.email}
```
